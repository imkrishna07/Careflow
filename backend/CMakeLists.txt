cmake_minimum_required(VERSION 3.16)
project(HospitalEmergency LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SQLite3 REQUIRED)
add_executable(hospital-api src/main.cpp src/MaxHeap.cpp src/Database.cpp)
target_include_directories(hospital-api PRIVATE include)
target_link_libraries(hospital-api PRIVATE SQLite::SQLite3)
add_executable(heap-tests tests/heap_tests.cpp src/MaxHeap.cpp)
target_include_directories(heap-tests PRIVATE include)
enable_testing()
add_test(NAME heap-tests COMMAND heap-tests)

add_executable(database-tests tests/database_tests.cpp src/Database.cpp)
target_include_directories(database-tests PRIVATE include)
target_link_libraries(database-tests PRIVATE SQLite::SQLite3)
add_test(NAME database-tests COMMAND database-tests)
